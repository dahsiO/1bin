Location : toString :
=====================

@Override
public String toString() {
	String location = "Location " ;
	if (/* Test si la location est terminée */)
		location = location + "terminée :" ;
	else
		location = location + "en cours :" ;
	int i=1 ;
	/* Parcours de tous les jeux contenus dans la location */ {
		location = location + '\n' + i+") "+ jeu;
		i++ ;
	}
	return location ;
}

	affilie = affilie + '\n'+"Liste des locations passées :" +'\n'+"-----------------------------" ;

		affilie = affilie + '\n'+ "Pas encore de location passée" ;

		/* Parcours des locations passées de l'affilié */ {
